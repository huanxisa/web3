#### JWT工具（utils/jwt.go）
```go
package utils

import (
    "time"
    "github.com/golang-jwt/jwt/v5"
)

var jwtSecret = []byte("your-secret-key") // 实际使用时从配置文件读取

type Claims struct {
    UserID uint `json:"user_id"`
    Email  string `json:"email"`
    jwt.RegisteredClaims
}

// GenerateToken 生成JWT Token
func GenerateToken(userID uint, email string) (string, error) {
    claims := Claims{
        UserID: userID,
        Email:  email,
        RegisteredClaims: jwt.RegisteredClaims{
            ExpiresAt: jwt.NewNumericDate(time.Now().Add(24 * time.Hour)),
            IssuedAt:  jwt.NewNumericDate(time.Now()),
        },
    }
    
    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    return token.SignedString(jwtSecret)
}

// ParseToken 解析JWT Token
func ParseToken(tokenString string) (*Claims, error) {
    token, err := jwt.ParseWithClaims(tokenString, &Claims{}, func(token *jwt.Token) (interface{}, error) {
        return jwtSecret, nil
    })
    
    if claims, ok := token.Claims.(*Claims); ok && token.Valid {
        return claims, nil
    }
    
    return nil, err
}
```

#### JWT认证中间件（middleware/auth.go）
```go
package middleware

import (
    "strings"
    "net/http"
    "github.com/gin-gonic/gin"
    "your-project/utils"
)

func AuthMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        // 从Header获取token
        authHeader := c.GetHeader("Authorization")
        if authHeader == "" {
            c.JSON(http.StatusUnauthorized, gin.H{"code": 401, "message": "请提供Token"})
            c.Abort()
            return
        }
        
        // 解析Bearer Token
        parts := strings.SplitN(authHeader, " ", 2)
        if !(len(parts) == 2 && parts[0] == "Bearer") {
            c.JSON(http.StatusUnauthorized, gin.H{"code": 401, "message": "Token格式错误"})
            c.Abort()
            return
        }
        
        // 验证Token
        claims, err := utils.ParseToken(parts[1])
        if err != nil {
            c.JSON(http.StatusUnauthorized, gin.H{"code": 401, "message": "Token无效或已过期"})
            c.Abort()
            return
        }
        
        // 将用户信息存储到上下文
        c.Set("user_id", claims.UserID)
        c.Set("email", claims.Email)
        c.Next()
    }
}
```

```go
package services

import (
    "context"
    "log"
    "math/big"
    "time"
    
    "github.com/ethereum/go-ethereum"
    "github.com/ethereum/go-ethereum/accounts/abi"
    "github.com/ethereum/go-ethereum/common"
    "github.com/ethereum/go-ethereum/core/types"
    "github.com/ethereum/go-ethereum/crypto"
    "github.com/ethereum/go-ethereum/ethclient"
)

type ListenerService struct {
    client *ethclient.Client
    abi    abi.ABI
}

func NewListenerService(rpcURL string, erc20ABI abi.ABI) (*ListenerService, error) {
    client, err := ethclient.Dial(rpcURL)
    if err != nil {
        return nil, err
    }
    
    return &ListenerService{
        client: client,
        abi:    erc20ABI,
    }, nil
}

// SubscribeTransfers 订阅Transfer事件
func (s *ListenerService) SubscribeTransfers(tokenAddress common.Address) {
    // Transfer事件的Topic
    transferEventSignature := []byte("Transfer(address,address,uint256)")
    transferTopic := crypto.Keccak256Hash(transferEventSignature)
    
    query := ethereum.FilterQuery{
        Addresses: []common.Address{tokenAddress},
        Topics:    [][]common.Hash{{transferTopic}},
    }
    
    logs := make(chan types.Log)
    ctx := context.Background()
    
    // 订阅
    sub, err := s.client.SubscribeFilterLogs(ctx, query, logs)
    if err != nil {
        log.Printf("订阅失败: %v", err)
        // 实现重连逻辑
        s.reconnect(tokenAddress)
        return
    }
    
    log.Printf("开始监听代币 %s 的Transfer事件", tokenAddress.Hex())
    
    for {
        select {
        case err := <-sub.Err():
            log.Printf("订阅错误: %v，尝试重连...", err)
            s.reconnect(tokenAddress)
            return
            
        case vLog := <-logs:
            // 解析事件
            s.handleTransferLog(vLog, tokenAddress)
        }
    }
}

// handleTransferLog 处理Transfer日志
func (s *ListenerService) handleTransferLog(vLog types.Log, tokenAddress common.Address) {
    // Transfer事件有3个参数：from (indexed), to (indexed), value
    // indexed参数在Topics中，非indexed参数在Data中
    
    if len(vLog.Topics) < 3 {
        log.Printf("日志Topics数量不足: %d", len(vLog.Topics))
        return
    }
    
    // 解析from和to（去掉前12个字节的0，只保留后20个字节地址）
    from := common.HexToAddress(vLog.Topics[1].Hex())
    to := common.HexToAddress(vLog.Topics[2].Hex())
    
    // 解析amount
    amount := new(big.Int).SetBytes(vLog.Data)
    
    // 获取区块时间戳
    block, err := s.client.BlockByNumber(context.Background(), big.NewInt(int64(vLog.BlockNumber)))
    if err != nil {
        log.Printf("获取区块失败: %v", err)
        return
    }
    
    // 存储到数据库
    transfer := &Transfer{
        TokenAddress:   tokenAddress.Hex(),
        FromAddress:    from.Hex(),
        ToAddress:      to.Hex(),
        Amount:         amount.String(),
        BlockNumber:    vLog.BlockNumber,
        TxHash:         vLog.TxHash.Hex(),
        BlockTimestamp: int64(block.Time()),
    }
    
    // 保存（这里简化了，实际应该调用repository层）
    log.Printf("Transfer: %s -> %s, amount: %s, block: %d",
        from.Hex(), to.Hex(), amount.String(), vLog.BlockNumber)
    
    // TODO: 保存到数据库
}

// reconnect 重连逻辑（指数退避）
func (s *ListenerService) reconnect(tokenAddress common.Address) {
    backoff := time.Second
    maxBackoff := time.Minute * 5
    
    for {
        log.Printf("等待 %v 后重连...", backoff)
        time.Sleep(backoff)
        
        // 尝试重新订阅
        go s.SubscribeTransfers(tokenAddress)
        
        // 指数退避
        backoff *= 2
        if backoff > maxBackoff {
            backoff = maxBackoff
        }
        
        return // 启动新的goroutine后退出当前
    }
}
```

### 核心代码示例

#### 数据同步服务（services/sync_service.go）
```go
package services

import (
    "context"
    "log"
    "math/big"
    "time"
    
    "github.com/ethereum/go-ethereum/ethclient"
    "github.com/ethereum/go-ethereum/core/types"
)

type SyncService struct {
    client *ethclient.Client
    blockRepo *BlockRepository
    txRepo *TxRepository
    addressRepo *AddressRepository
}

// Start 启动同步服务
func (s *SyncService) Start() {
    // 订阅新区块
    go s.subscribeNewBlocks()
    
    // 定期同步遗漏的区块（fallback机制）
    go s.periodicSync()
}

// subscribeNewBlocks 订阅新区块
func (s *SyncService) subscribeNewBlocks() {
    headers := make(chan *types.Header)
    ctx := context.Background()
    
    sub, err := s.client.SubscribeNewHead(ctx, headers)
    if err != nil {
        log.Printf("订阅新区块失败: %v", err)
        // 实现重连
        time.Sleep(5 * time.Second)
        s.subscribeNewBlocks()
        return
    }
    
    log.Println("开始订阅新区块...")
    
    for {
        select {
        case err := <-sub.Err():
            log.Printf("订阅错误: %v，重连中...", err)
            time.Sleep(5 * time.Second)
            s.subscribeNewBlocks()
            return
            
        case header := <-headers:
            log.Printf("收到新区块: %d", header.Number.Uint64())
            s.processBlock(header.Number)
        }
    }
}

// processBlock 处理区块
func (s *SyncService) processBlock(blockNumber *big.Int) {
    ctx := context.Background()
    
    // 获取完整区块数据
    block, err := s.client.BlockByNumber(ctx, blockNumber)
    if err != nil {
        log.Printf("获取区块失败: %v", err)
        return
    }
    
    // 保存区块信息
    blockModel := &Block{
        Number:           block.Number().Uint64(),
        Hash:             block.Hash().Hex(),
        ParentHash:       block.ParentHash().Hex(),
        Timestamp:        int64(block.Time()),
        Miner:            block.Coinbase().Hex(),
        GasUsed:          block.GasUsed(),
        GasLimit:         block.GasLimit(),
        TransactionCount: len(block.Transactions()),
    }
    
    if err := s.blockRepo.Save(blockModel); err != nil {
        log.Printf("保存区块失败: %v", err)
        return
    }
    
    // 处理区块中的所有交易
    for _, tx := range block.Transactions() {
        s.processTransaction(tx, block)
    }
    
    log.Printf("区块 %d 同步完成，包含 %d 笔交易",
        block.Number().Uint64(), len(block.Transactions()))
}

// processTransaction 处理交易
func (s *SyncService) processTransaction(tx *types.Transaction, block *types.Block) {
    // 获取交易回执（包含status）
    receipt, err := s.client.TransactionReceipt(context.Background(), tx.Hash())
    if err != nil {
        log.Printf("获取交易回执失败: %v", err)
        return
    }
    
    // 获取发送方地址
    msg, err := tx.AsMessage(types.LatestSignerForChainID(tx.ChainId()), nil)
    if err != nil {
        log.Printf("解析交易失败: %v", err)
        return
    }
    
    // 构造交易模型
    txModel := &Transaction{
        Hash:           tx.Hash().Hex(),
        BlockNumber:    block.Number().Uint64(),
        FromAddress:    msg.From().Hex(),
        ToAddress:      "",
        Value:          tx.Value().String(),
        GasUsed:        receipt.GasUsed,
        GasPrice:       tx.GasPrice().String(),
        Nonce:          tx.Nonce(),
        Status:         receipt.Status,
        BlockTimestamp: int64(block.Time()),
    }
    
    if tx.To() != nil {
        txModel.ToAddress = tx.To().Hex()
    }
    
    // 保存交易
    if err := s.txRepo.Save(txModel); err != nil {
        log.Printf("保存交易失败: %v", err)
        return
    }
    
    // 更新地址统计
    s.updateAddressStats(msg.From().Hex(), block.Number().Uint64())
    if tx.To() != nil {
        s.updateAddressStats(tx.To().Hex(), block.Number().Uint64())
    }
}

// updateAddressStats 更新地址统计
func (s *SyncService) updateAddressStats(address string, blockNumber uint64) {
    // 查询最新余额
    balance, err := s.client.BalanceAt(context.Background(),
        common.HexToAddress(address), big.NewInt(int64(blockNumber)))
    if err != nil {
        log.Printf("查询余额失败: %v", err)
        return
    }
    
    // 更新或创建地址记录
    s.addressRepo.UpdateStats(address, balance.String(), blockNumber)
}
```
